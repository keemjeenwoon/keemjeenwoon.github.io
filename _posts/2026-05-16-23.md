<script>
(function() {
  const IMG_PATH = '/assets/images/plinko/';
  const IMGS = [
    IMG_PATH + 'plinko01.png',
    IMG_PATH + 'plinko02.png',
    IMG_PATH + 'plinko03.png',
    IMG_PATH + 'plinko04.png',
    IMG_PATH + 'plinko05.png',
    IMG_PATH + 'plinko06.png',
  ];

  const canvas = document.getElementById('plinko-c');
  const ctx = canvas.getContext('2d');
  const W = canvas.width, H = canvas.height;

  const PEG_R = 8;
  const BALL_R = 28;
  const FLOOR_Y = H - 30;
  const GRAVITY = 0.28;
  const BOUNCE = 0.38;
  const FRICTION = 0.98;

  let balls = [], pegs = [];
  let dropCount = 0, landCount = 0, firstWinner = null, nextImgIdx = 0;

  const noodleImgs = IMGS.map(src => {
    const img = new Image();
    img.src = src;
    return img;
  });

  function buildPegs() {
    pegs = [];
    const ROWS = 5, maxCols = 5;
    const colSpacing = 80, rowSpacing = 82;
    const startX = (W - (maxCols - 1) * colSpacing) / 2;
    const startY = 60;
    for (let r = 0; r < ROWS; r++) {
      const cols = r + 1;
      const offsetX = ((maxCols - cols) * colSpacing) / 2;
      for (let c = 0; c < cols; c++) {
        pegs.push({ x: startX + offsetX + c * colSpacing, y: startY + r * rowSpacing });
      }
    }
  }

  function makeBall(x) {
    dropCount++;
    document.getElementById('plinko-cnt').textContent = dropCount;
    const imgIdx = nextImgIdx % IMGS.length;
    nextImgIdx++;
    return { x, y: 28, vx: (Math.random() - 0.5) * 1.2, vy: 0, r: BALL_R, id: dropCount, imgIdx, landed: false };
  }

  window.plinkoDrop = function() {
    balls.push(makeBall(W / 2 + (Math.random() - 0.5) * 20));
  };
  window.plinkoDropMany = function() {
    for (let i = 0; i < 5; i++)
      setTimeout(() => balls.push(makeBall(W / 2 + (Math.random() - 0.5) * 40)), i * 140);
  };
  window.plinkoReset = function() {
    balls = [];
    dropCount = 0; landCount = 0; firstWinner = null; nextImgIdx = 0;
    document.getElementById('plinko-cnt').textContent = 0;
    document.getElementById('plinko-land').textContent = 0;
    document.getElementById('plinko-winner').textContent = '—';
  };

  function drawPeg(x, y) {
    ctx.beginPath();
    ctx.arc(x, y, PEG_R, 0, Math.PI * 2);
    ctx.fillStyle = '#000';
    ctx.fill();
  }

  function drawBall(b) {
    const img = noodleImgs[b.imgIdx];
    ctx.save();
    ctx.beginPath();
    ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
    ctx.clip();
    if (img.complete && img.naturalWidth > 0) {
      ctx.drawImage(img, b.x - b.r, b.y - b.r, b.r * 2, b.r * 2);
    } else {
      ctx.fillStyle = '#f5ead0';
      ctx.fill();
    }
    ctx.restore();
    ctx.fillStyle = '#333';
    ctx.font = 'bold 10px monospace';
    ctx.textAlign = 'center';
    ctx.textBaseline = 'top';
    ctx.fillText('#' + b.id, b.x, b.y + b.r + 3);
  }

  function update(b) {
    if (b.landed) return;
    b.vy += GRAVITY;
    b.vx *= FRICTION;
    b.x += b.vx;
    b.y += b.vy;
    for (const p of pegs) {
      const dx = b.x - p.x, dy = b.y - p.y;
      const dist = Math.sqrt(dx * dx + dy * dy);
      const minD = b.r + PEG_R;
      if (dist < minD && dist > 0.01) {
        const nx = dx / dist, ny = dy / dist;
        b.x = p.x + nx * minD;
        b.y = p.y + ny * minD;
        const dot = b.vx * nx + b.vy * ny;
        b.vx = (b.vx - 2 * dot * nx) * BOUNCE + (Math.random() - 0.5) * 0.5;
        b.vy = (b.vy - 2 * dot * ny) * BOUNCE;
      }
    }
    if (b.y + b.r >= FLOOR_Y) {
      b.y = FLOOR_Y - b.r;
      b.vy *= -0.3;
      b.vx *= 0.7;
      if (Math.abs(b.vy) < 0.5) {
        b.landed = true;
        landCount++;
        document.getElementById('plinko-land').textContent = landCount;
        if (!firstWinner) {
          firstWinner = b.id;
          document.getElementById('plinko-winner').textContent = '공 #' + firstWinner;
        }
      }
    }
    if (b.x - b.r < 10) { b.x = 10 + b.r; b.vx = Math.abs(b.vx); }
    if (b.x + b.r > W - 10) { b.x = W - 10 - b.r; b.vx = -Math.abs(b.vx); }
  }

  function loop() {
    ctx.clearRect(0, 0, W, H);
    pegs.forEach(p => drawPeg(p.x, p.y));
    balls.forEach(b => { update(b); drawBall(b); });
    requestAnimationFrame(loop);
  }

  buildPegs();
  loop();
})();
</script>
