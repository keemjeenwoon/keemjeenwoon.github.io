const overlay = document.getElementById('modalOverlay');
const flipCard = document.getElementById('flipCard');
const cards = document.querySelectorAll('.cards li');
let currentClickedImg = null;

// 앨범 클릭 이벤트
cards.forEach(card => {
  card.addEventListener('click', function(e) {
    const albumId = this.getAttribute('data-album');
    const album = albumData[albumId];
    
    if (album) {
      // 클릭한 img의 실제 렌더링된 위치와 크기 가져오기
      const img = this.querySelector('img');
      const imgRect = img.getBoundingClientRect();
      
      // 현재 클릭한 이미지 저장하고 숨기기
      currentClickedImg = img;
      img.classList.add('hidden');
      
      // 플립 카드를 이미지의 정확한 위치와 크기로 설정
      flipCard.style.left = imgRect.left + 'px';
      flipCard.style.top = imgRect.top + 'px';
      flipCard.style.width = imgRect.width + 'px';
      flipCard.style.height = imgRect.height + 'px';
      
      // 데이터 설정
      document.getElementById('flipCover').src = `/assets/images/coverflow/album${albumId}.jpg`;
      document.getElementById('flipTitle').textContent = album.title;
      document.getElementById('flipArtist').textContent = album.artist;
      document.getElementById('flipReview').textContent = album.review;
      
      // 활성화
      overlay.classList.add('active');
      flipCard.classList.add('active');
      
      // 약간의 딜레이 후 뒤집기
      setTimeout(() => {
        flipCard.classList.add('flipped');
      }, 50);
    }
  });
});

// 배경 클릭 시 닫기
overlay.addEventListener('click', function() {
  flipCard.classList.remove('flipped');
  setTimeout(() => {
    flipCard.classList.remove('active');
    overlay.classList.remove('active');
    
    // 숨겼던 이미지 다시 보이기
    if (currentClickedImg) {
      currentClickedImg.classList.remove('hidden');
      currentClickedImg = null;
    }
  }, 600);
});
</script>
